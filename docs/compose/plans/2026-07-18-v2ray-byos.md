# V2Ray/Xray BYOS (Bring Your Own Server) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use compose:subagent (recommended) or compose:execute to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace VPN Gate (SoftEther) as the server source with V2Ray/Xray BYOS — users paste subscription URLs or vmess/vless/trojan/ss:// URIs to import their own servers, then connect smoothly via the axevpn_flutter V2Ray engine.

**Architecture:** Add a V2Ray service layer mirroring the existing OpenVPN pattern (V2RayService wraps the plugin's `V2Ray` class). Add a config parser that converts vmess/vless/trojan/ss URIs into Xray-core JSON. Add an import server screen with URL paste, clipboard detection, and QR code scanning. Extend VPNProvider to branch by protocol (OpenVPN vs V2Ray).

**Tech Stack:** Flutter, axevpn_flutter (V2Ray sub-plugin), Xray-core JSON format, SharedPreferences for persistence, base64 decoding, URI parsing.

```dart
// lib/services/v2ray_config_parser.dart
import 'dart:convert';

enum V2RayProtocol { vmess, vless, trojan, shadowsocks, unknown }

class ParsedV2RayConfig {
  final V2RayProtocol protocol;
  final String name;
  final String address;
  final int port;
  final String jsonConfig;
  final Map<String, dynamic> metadata;

  const ParsedV2RayConfig({
    required this.protocol,
    required this.name,
    required this.address,
    required this.port,
    required this.jsonConfig,
    this.metadata = const {},
  });
}

class V2RayConfigParser {
  static ParsedV2RayConfig? parse(String input) {
    input = input.trim();
    if (input.startsWith('vmess://')) return _parseVmess(input.substring(8));
    if (input.startsWith('vless://')) return _parseVless(input.substring(8));
    if (input.startsWith('trojan://')) return _parseTrojan(input.substring(9));
    if (input.startsWith('ss://')) return _parseShadowsocks(input.substring(5));
    if (input.startsWith('{')) return _parseRawJson(input);
    return null;
  }

  static List<ParsedV2RayConfig> parseSubscription(String response) {
    try {
      final decoded = utf8.decode(base64Url.decode(response));
      final lines = decoded.split('\n').where((l) => l.trim().isNotEmpty).toList();
      return lines.map((l) => parse(l)).whereType<ParsedV2RayConfig>().toList();
    } catch (_) {
      return [];
    }
  }

  static ParsedV2RayConfig? _parseVmess(String encoded) {
    try {
      final jsonStr = utf8.decode(base64Url.decode(encoded));
      final json = jsonDecode(jsonStr) as Map<String, dynamic>;
      final ps = json['ps'] ?? json['add'] ?? 'VMess Server';
      final add = json['add'] ?? '';
      final port = int.tryParse('${json['port']}') ?? 443;
      final uuid = json['id'] ?? '';
      final aid = json['aid'] ?? '0';
      final net = json['net'] ?? 'tcp';
      final type = json['type'] ?? 'none';
      final host = json['host'] ?? '';
      final path = json['path'] ?? '';
      final tls = json['tls'] ?? '';
      final sni = json['sni'] ?? host;

      final streamSettings = <String, dynamic>{
        'network': net,
      };

      if (net == 'ws') {
        streamSettings['wsSettings'] = {
          'path': path,
          if (host.isNotEmpty) 'headers': {'Host': host},
        };
      } else if (net == 'grpc') {
        streamSettings['grpcSettings'] = {
          'serviceName': path,
        };
      } else if (net == 'h2' || net == 'http') {
        streamSettings['httpSettings'] = {
          'path': path,
          if (host.isNotEmpty) 'host': [host],
        };
      }

      if (tls == 'tls') {
        streamSettings['security'] = 'tls';
        streamSettings['tlsSettings'] = {
          'serverName': sni,
          'allowInsecure': true,
        };
      }

      final config = _buildFullConfig('vmess', {
        'vnext': [{
          'address': add,
          'port': port,
          'users': [{
            'id': uuid,
            'alterId': int.tryParse('$aid') ?? 0,
            'security': 'auto',
          }],
        }],
      }, streamSettings);

      return ParsedV2RayConfig(
        protocol: V2RayProtocol.vmess,
        name: ps.toString(),
        address: add,
        port: port,
        jsonConfig: jsonEncode(config),
        metadata: {'uuid': uuid, 'alterId': aid, 'network': net, 'tls': tls},
      );
    } catch (_) {
      return null;
    }
  }

  static ParsedV2RayConfig? _parseVless(String input) {
    try {
      final atIndex = input.indexOf('@');
      if (atIndex == -1) return null;

      final userInfo = input.substring(0, atIndex);
      final rest = input.substring(atIndex + 1);

      final questionIndex = rest.indexOf('?');
      final fragmentIndex = rest.indexOf('#');
      String hostPort;
      String queryParams = '';
      String name = 'VLESS Server';

      if (fragmentIndex != -1) {
        name = Uri.decodeComponent(rest.substring(fragmentIndex + 1));
        hostPort = questionIndex != -1 && questionIndex < fragmentIndex
            ? rest.substring(0, questionIndex)
            : rest.substring(0, fragmentIndex);
      } else {
        hostPort = questionIndex != -1 ? rest.substring(0, questionIndex) : rest;
      }

      if (questionIndex != -1) {
        final end = fragmentIndex != -1 ? fragmentIndex : rest.length;
        queryParams = rest.substring(questionIndex + 1, end);
      }

      final colonIndex = hostPort.lastIndexOf(':');
      if (colonIndex == -1) return null;
      final address = hostPort.substring(0, colonIndex);
      final port = int.tryParse(hostPort.substring(colonIndex + 1)) ?? 443;

      final params = Uri.splitQueryString(queryParams);
      final security = params['security'] ?? 'none';
      final network = params['type'] ?? 'tcp';
      final sni = params['sni'] ?? params['host'] ?? '';
      final path = params['path'] ?? '';
      final host = params['host'] ?? '';
      final flow = params['flow'] ?? '';

      final streamSettings = <String, dynamic>{'network': network};

      if (network == 'ws') {
        streamSettings['wsSettings'] = {
          'path': path,
          if (host.isNotEmpty) 'headers': {'Host': host},
        };
      } else if (network == 'grpc') {
        streamSettings['grpcSettings'] = {'serviceName': params['serviceName'] ?? ''};
      } else if (network == 'h2' || network == 'http') {
        streamSettings['httpSettings'] = {
          'path': path,
          if (host.isNotEmpty) 'host': [host],
        };
      }

      if (security == 'tls') {
        streamSettings['security'] = 'tls';
        streamSettings['tlsSettings'] = {
          'serverName': sni,
          'allowInsecure': true,
        };
      } else if (security == 'reality') {
        streamSettings['security'] = 'reality';
        streamSettings['realitySettings'] = {
          'serverName': sni,
          'fingerprint': params['fp'] ?? 'chrome',
          'publicKey': params['pbk'] ?? '',
          'shortId': params['sid'] ?? '',
          'spiderX': params['spx'] ?? '',
        };
      }

      final user = <String, dynamic>{
        'id': userInfo,
        'encryption': params['encryption'] ?? 'none',
      };
      if (flow.isNotEmpty) user['flow'] = flow;

      final config = _buildFullConfig('vless', {
        'vnext': [{
          'address': address,
          'port': port,
          'users': [user],
        }],
      }, streamSettings);

      return ParsedV2RayConfig(
        protocol: V2RayProtocol.vless,
        name: name,
        address: address,
        port: port,
        jsonConfig: jsonEncode(config),
        metadata: {'uuid': userInfo, 'security': security, 'network': network},
      );
    } catch (_) {
      return null;
    }
  }

  static ParsedV2RayConfig? _parseTrojan(String input) {
    try {
      final atIndex = input.indexOf('@');
      if (atIndex == -1) return null;
      final password = Uri.decodeComponent(input.substring(0, atIndex));
      final rest = input.substring(atIndex + 1);

      final fragmentIndex = rest.indexOf('#');
      final questionIndex = rest.indexOf('?');
      String hostPort;
      String name = 'Trojan Server';
      String queryParams = '';

      if (fragmentIndex != -1) {
        name = Uri.decodeComponent(rest.substring(fragmentIndex + 1));
        hostPort = rest.substring(0, fragmentIndex);
      } else {
        hostPort = rest;
      }
      if (questionIndex != -1 && questionIndex < hostPort.length) {
        final qEnd = fragmentIndex != -1 ? fragmentIndex : rest.length;
        queryParams = rest.substring(questionIndex + 1, qEnd);
        hostPort = hostPort.substring(0, questionIndex);
      }

      final colonIndex = hostPort.lastIndexOf(':');
      if (colonIndex == -1) return null;
      final address = hostPort.substring(0, colonIndex);
      final port = int.tryParse(hostPort.substring(colonIndex + 1)) ?? 443;

      final params = Uri.splitQueryString(queryParams);
      final security = params['security'] ?? 'tls';
      final sni = params['sni'] ?? params['host'] ?? address;
      final network = params['type'] ?? 'tcp';

      final streamSettings = <String, dynamic>{'network': network};
      if (security == 'tls') {
        streamSettings['security'] = 'tls';
        streamSettings['tlsSettings'] = {
          'serverName': sni,
          'allowInsecure': true,
        };
      }
      if (network == 'ws') {
        streamSettings['wsSettings'] = {
          'path': params['path'] ?? '',
          if ((params['host'] ?? '').isNotEmpty) 'headers': {'Host': params['host']},
        };
      }

      final config = _buildFullConfig('trojan', {
        'servers': [{
          'address': address,
          'port': port,
          'password': password,
        }],
      }, streamSettings);

      return ParsedV2RayConfig(
        protocol: V2RayProtocol.trojan,
        name: name,
        address: address,
        port: port,
        jsonConfig: jsonEncode(config),
        metadata: {'password': password, 'security': security},
      );
    } catch (_) {
      return null;
    }
  }

  static ParsedV2RayConfig? _parseShadowsocks(String input) {
    try {
      String decoded;
      final hashIndex = input.indexOf('#');
      String name = 'Shadowsocks Server';
      String core = input;

      if (hashIndex != -1) {
        name = Uri.decodeComponent(input.substring(hashIndex + 1));
        core = input.substring(0, hashIndex);
      }

      final atIndex = core.indexOf('@');
      if (atIndex != -1) {
        decoded = Uri.decodeComponent(core);
      } else {
        decoded = utf8.decode(base64Url.normalize(base64Url.decode(core)));
      }

      final colonIndex = decoded.lastIndexOf(':');
      if (colonIndex == -1) return null;
      final address = decoded.substring(0, colonIndex);
      final rest = decoded.substring(colonIndex + 1);
      final spaceIndex = rest.indexOf(' ');
      final port = int.tryParse(rest.substring(0, spaceIndex != -1 ? spaceIndex : rest.length)) ?? 443;
      final method = spaceIndex != -1 ? rest.substring(spaceIndex + 1).trim() : 'aes-256-gcm';

      final config = _buildFullConfig('shadowsocks', {
        'servers': [{
          'address': address,
          'port': port,
          'method': method,
          'password': '',
        }],
      }, {});

      return ParsedV2RayConfig(
        protocol: V2RayProtocol.shadowsocks,
        name: name,
        address: address,
        port: port,
        jsonConfig: jsonEncode(config),
        metadata: {'method': method},
      );
    } catch (_) {
      return null;
    }
  }

  static ParsedV2RayConfig? _parseRawJson(String jsonStr) {
    try {
      final json = jsonDecode(jsonStr) as Map<String, dynamic>;
      final outbounds = json['outbounds'] as List?;
      if (outbounds == null || outbounds.isEmpty) return null;
      final outbound = outbounds[0] as Map<String, dynamic>;
      final protocol = outbound['protocol'] as String? ?? 'unknown';
      final settings = outbound['settings'] as Map<String, dynamic>? ?? {};

      String address = '';
      int port = 0;
      String name = 'V2Ray Server';

      final vnext = settings['vnext'] as List?;
      if (vnext != null && vnext.isNotEmpty) {
        final v = vnext[0] as Map<String, dynamic>;
        address = v['address'] ?? '';
        port = v['port'] ?? 0;
      }
      final servers = settings['servers'] as List?;
      if (servers != null && servers.isNotEmpty) {
        final s = servers[0] as Map<String, dynamic>;
        address = s['address'] ?? '';
        port = s['port'] ?? 0;
      }

      final protoEnum = V2RayProtocol.values.firstWhere(
        (p) => p.name == protocol,
        orElse: () => V2RayProtocol.unknown,
      );

      return ParsedV2RayConfig(
        protocol: protoEnum,
        name: name,
        address: address,
        port: port,
        jsonConfig: jsonStr,
      );
    } catch (_) {
      return null;
    }
  }

  static Map<String, dynamic> _buildFullConfig(
    String protocol,
    Map<String, dynamic> outboundSettings,
    Map<String, dynamic> streamSettings,
  ) {
    final outbound = <String, dynamic>{
      'protocol': protocol,
      'settings': outboundSettings,
    };
    if (streamSettings.isNotEmpty) {
      outbound['streamSettings'] = streamSettings;
    }
    return {
      'log': {'loglevel': 'warning'},
      'dns': {'servers': ['1.1.1.1', '8.8.8.8']},
      'inbounds': [
        {
          'port': 10808,
          'protocol': 'socks',
          'settings': {'auth': 'noauth', 'udp': true},
        },
      ],
      'outbounds': [outbound],
    };
  }
}
```

```dart
// lib/services/v2ray_service.dart
import 'dart:async';
import 'package:axevpn_flutter/v2ray_flutter.dart' as axe;

enum V2RayVpnStage { disconnected, connecting, connected, disconnecting, error }

class V2RayService {
  late axe.V2Ray _v2ray;
  final _stageController = StreamController<V2RayVpnStage>.broadcast();
  V2RayVpnStage _currentStage = V2RayVpnStage.disconnected;
  bool _initialized = false;

  Stream<V2RayVpnStage> get stageStream => _stageController.stream;
  V2RayVpnStage get currentStage => _currentStage;

  String get stageLabel {
    switch (_currentStage) {
      case V2RayVpnStage.disconnected:
        return 'Disconnected';
      case V2RayVpnStage.connecting:
        return 'Connecting...';
      case V2RayVpnStage.connected:
        return 'Connected';
      case V2RayVpnStage.disconnecting:
        return 'Disconnecting...';
      case V2RayVpnStage.error:
        return 'Error';
    }
  }

  Future<void> initialize() async {
    if (_initialized) return;

    _v2ray = axe.V2Ray(
      onVpnStatusChanged: (status) {},
      onVpnStageChanged: (stage, raw) {
        final mapped = _mapStage(stage);
        _currentStage = mapped;
        _stageController.add(mapped);
      },
    );

    await _v2ray.initialize();
    _initialized = true;
  }

  Future<void> connect(String configJson, String name, {axe.V2RaySubProtocol subProtocol = axe.V2RaySubProtocol.vless, List<String>? bypassPackages}) async {
    if (!_initialized) await initialize();

    _currentStage = V2RayVpnStage.connecting;
    _stageController.add(V2RayVpnStage.connecting);

    try {
      await _v2ray.connect(configJson, name, subProtocol: subProtocol, bypassPackages: bypassPackages);
    } catch (e) {
      _currentStage = V2RayVpnStage.error;
      _stageController.add(V2RayVpnStage.error);
    }
  }

  Future<void> disconnect() async {
    _currentStage = V2RayVpnStage.disconnecting;
    _stageController.add(V2RayVpnStage.disconnecting);
    try {
      await _v2ray.disconnect();
    } catch (_) {}
  }

  Future<axe.V2RayStatus> status() async => _v2ray.status();

  V2RayVpnStage stage() => _currentStage;
  bool isConnected() => _currentStage == V2RayVpnStage.connected;

  V2RayVpnStage _mapStage(axe.V2RayStage stage) {
    switch (stage) {
      case axe.V2RayStage.connected:
        return V2RayVpnStage.connected;
      case axe.V2RayStage.connecting:
      case axe.V2RayStage.preparing:
        return V2RayVpnStage.connecting;
      case axe.V2RayStage.disconnecting:
        return V2RayVpnStage.disconnecting;
      case axe.V2RayStage.disconnected:
      case axe.V2RayStage.denied:
      case axe.V2RayStage.error:
      case axe.V2RayStage.unknown:
        return V2RayVpnStage.disconnected;
    }
  }

  void dispose() {
    _stageController.close();
  }
}
```

```dart
import '../services/v2ray_config_parser.dart';

class ImportedServer {
  final String id;
  final String name;
  final String address;
  final int port;
  final V2RayProtocol protocol;
  final String configJson;
  final DateTime importedAt;
  final bool isFavorite;

  const ImportedServer({
    required this.id,
    required this.name,
    required this.address,
    required this.port,
    required this.protocol,
    required this.configJson,
    required this.importedAt,
    this.isFavorite = false,
  });

  Map<String, dynamic> toJson() => {
    'id': id,
    'name': name,
    'address': address,
    'port': port,
    'protocol': protocol.name,
    'configJson': configJson,
    'importedAt': importedAt.toIso8601String(),
    'isFavorite': isFavorite,
  };

  factory ImportedServer.fromJson(Map<String, dynamic> json) => ImportedServer(
    id: json['id'],
    name: json['name'],
    address: json['address'],
    port: json['port'],
    protocol: V2RayProtocol.values.firstWhere((p) => p.name == json['protocol'], orElse: () => V2RayProtocol.unknown),
    configJson: json['configJson'],
    importedAt: DateTime.parse(json['importedAt']),
    isFavorite: json['isFavorite'] ?? false,
  );

  ImportedServer copyWith({String? name, bool? isFavorite}) => ImportedServer(
    id: id,
    name: name ?? this.name,
    address: address,
    port: port,
    protocol: protocol,
    configJson: configJson,
    importedAt: importedAt,
    isFavorite: isFavorite ?? this.isFavorite,
  );
}
```

```dart
// lib/services/server_storage_service.dart
import 'dart:convert';
import 'package:shared_preferences/shared_preferences.dart';
import '../models/imported_server.dart';

class ServerStorageService {
  static const String _key = 'imported_servers';

  Future<List<ImportedServer>> loadServers() async {
    final prefs = await SharedPreferences.getInstance();
    final json = prefs.getString(_key);
    if (json == null) return [];
    final list = jsonDecode(json) as List;
    return list.map((e) => ImportedServer.fromJson(e)).toList();
  }

  Future<void> saveServers(List<ImportedServer> servers) async {
    final prefs = await SharedPreferences.getInstance();
    final json = servers.map((s) => s.toJson()).toList();
    await prefs.setString(_key, jsonEncode(json));
  }

  Future<void> addServer(ImportedServer server) async {
    final servers = await loadServers();
    servers.add(server);
    await saveServers(servers);
  }

  Future<void> removeServer(String id) async {
    final servers = await loadServers();
    servers.removeWhere((s) => s.id == id);
    await saveServers(servers);
  }

  Future<void> toggleFavorite(String id) async {
    final servers = await loadServers();
    final index = servers.indexWhere((s) => s.id == id);
    if (index != -1) {
      servers[index] = servers[index].copyWith(isFavorite: !servers[index].isFavorite);
      await saveServers(servers);
    }
  }
}
```

```dart
Future<void> connectV2Ray(ImportedServer server) async {
  _status = VPNStatus.connecting;
  notifyListeners();

  try {
    await _v2rayService!.initialize();
    await _v2rayService!.connect(
      server.configJson,
      server.name,
      subProtocol: _v2raySubProtocolFor(server.protocol),
    );
    _storageService.setLastServerId(server.id);
  } catch (e) {
    _status = VPNStatus.disconnected;
    notifyListeners();
  }
}

    case V2RayProtocol.unknown: return axe.V2RaySubProtocol.unknown;
```

```dart
Future<bool> importConfig(String rawInput) async {
  final parsed = V2RayConfigParser.parse(rawInput);
  if (parsed == null) return false;

  final server = ImportedServer(
    id: 'v2ray_${DateTime.now().millisecondsSinceEpoch}',
    name: parsed.name,
    address: parsed.address,
    port: parsed.port,
    protocol: parsed.protocol,
    configJson: parsed.jsonConfig,
    importedAt: DateTime.now(),
  );

  _importedServers.add(server);
  await _serverStorage.saveServers(_importedServers);
  notifyListeners();
  return true;
}

Future<bool> importSubscription(String url) async {
  try {
    final response = await http.get(Uri.parse(url));
    if (response.statusCode != 200) return false;

    final configs = V2RayConfigParser.parseSubscription(response.body);
    if (configs.isEmpty) return false;

    for (final config in configs) {
      final server = ImportedServer(
        id: 'v2ray_${DateTime.now().millisecondsSinceEpoch}_${config.address}',
        name: config.name,
        address: config.address,
        port: config.port,
        protocol: config.protocol,
        configJson: config.jsonConfig,
        importedAt: DateTime.now(),
      );
      _importedServers.add(server);
    }
    await _serverStorage.saveServers(_importedServers);
    notifyListeners();
    return true;
  } catch (_) {
    return false;
  }
}

Future<void> removeImportedServer(String id) async {
  _importedServers.removeWhere((s) => s.id == id);
  await _serverStorage.saveServers(_importedServers);
  notifyListeners();
}
```

```dart
// lib/screens/import_server_screen.dart
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:provider/provider.dart';
import '../providers/vpn_provider.dart';
import '../models/imported_server.dart';

class ImportServerScreen extends StatefulWidget {
  const ImportServerScreen({super.key});

  @override
  State<ImportServerScreen> createState() => _ImportServerScreenState();
}

class _ImportServerScreenState extends State<ImportServerScreen> {
  final _urlController = TextEditingController();
  final _configController = TextEditingController();
  bool _isSubscription = false;
  bool _isLoading = false;

  @override
  void dispose() {
    _urlController.dispose();
    _configController.dispose();
    super.dispose();
  }

  Future<void> _importFromClipboard() async {
    final data = await Clipboard.getData(Clipboard.kTextPlain);
    if (data?.text == null || data!.text!.isEmpty) {
      _showSnackBar('Clipboard is empty');
      return;
    }
    await _importConfig(data.text!);
  }

  Future<void> _importConfig(String input) async {
    setState(() => _isLoading = true);
    final vpn = context.read<VPNProvider>();
    final success = await vpn.importConfig(input);
    setState(() => _isLoading = false);

    if (success && mounted) {
      _showSnackBar('Server imported successfully');
      Navigator.pop(context);
    } else if (mounted) {
      _showSnackBar('Invalid server configuration');
    }
  }

  Future<void> _importSubscription() async {
    final url = _urlController.text.trim();
    if (url.isEmpty) {
      _showSnackBar('Enter a subscription URL');
      return;
    }
    setState(() => _isLoading = true);
    final vpn = context.read<VPNProvider>();
    final success = await vpn.importSubscription(url);
    setState(() => _isLoading = false);

    if (success && mounted) {
      _showSnackBar('Subscription imported');
      Navigator.pop(context);
    } else if (mounted) {
      _showSnackBar('Failed to import subscription');
    }
  }

  void _showSnackBar(String message) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message), duration: const Duration(seconds: 2)),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Add Server')),
      body: Padding(
        padding: const EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            // Clipboard paste button
            ElevatedButton.icon(
              onPressed: _isLoading ? null : _importFromClipboard,
              icon: const Icon(Icons.content_paste),
              label: const Text('Import from Clipboard'),
              style: ElevatedButton.styleFrom(padding: const EdgeInsets.all(16)),
            ),
            const SizedBox(height: 16),
            const Divider(),
            const SizedBox(height: 16),
            // Toggle between URL and config
            SegmentedButton<bool>(
              segments: const [
                ButtonSegment(value: true, label: Text('Subscription URL'), icon: Icon(Icons.link)),
                ButtonSegment(value: false, label: Text('Server Config'), icon: Icon(Icons.code)),
              ],
              selected: {_isSubscription},
              onSelectionChanged: (v) => setState(() => _isSubscription = v.first),
            ),
            const SizedBox(height: 16),
            if (_isSubscription) ...[
              TextField(
                controller: _urlController,
                decoration: const InputDecoration(
                  labelText: 'Subscription URL',
                  hintText: 'https://example.com/sub',
                  border: OutlineInputBorder(),
                ),
                keyboardType: TextInputType.url,
              ),
              const SizedBox(height: 16),
              ElevatedButton(
                onPressed: _isLoading ? null : _importSubscription,
                child: _isLoading
                    ? const SizedBox(height: 20, width: 20, child: CircularProgressIndicator(strokeWidth: 2))
                    : const Text('Import Subscription'),
              ),
            ] else ...[
              Expanded(
                child: TextField(
                  controller: _configController,
                  decoration: const InputDecoration(
                    labelText: 'Server Config (vmess://, vless://, trojan://, ss://)',
                    hintText: 'vless://uuid@host:port?...',
                    border: OutlineInputBorder(),
                    alignLabelWithHint: true,
                  ),
                  maxLines: null,
                  expands: true,
                  textAlignVertical: TextAlignVertical.top,
                ),
              ),
              const SizedBox(height: 16),
              ElevatedButton(
                onPressed: _isLoading
                    ? null
                    : () => _importConfig(_configController.text.trim()),
                child: _isLoading
                    ? const SizedBox(height: 20, width: 20, child: CircularProgressIndicator(strokeWidth: 2))
                    : const Text('Import Config'),
              ),
            ],
          ],
        ),
      ),
    );
  }
}
```
